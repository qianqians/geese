//! 第三人称轨道摄像机（原神风格）。
//!
//! 摄像机围绕角色旋转，鼠标拖拽控制水平/垂直角度，
//! 滚轮控制距离。平滑跟随角色移动。
//!
//! 初始轨道参数由项目配置 `config/project.toml` 注入：
//! - `follow_offset` 派生初始距离 / 俯仰角 / 水平角
//! - `eye_height` 作为注视点高度（看向角色身体而非脚底）
//! - `follow_smooth` 跟随平滑系数
//! - `mouse_sensitivity` 旋转灵敏度

use cgmath::{Matrix4, Point3, Rad, Vector3};

/// 第三人称轨道摄像机控制器。
pub struct ThirdPersonCamera {
    /// 摄像机世界位置
    pub position: Point3<f32>,
    /// 视场角（度）
    pub fov: f32,
    /// 近裁剪面
    pub z_near: f32,
    /// 远裁剪面
    pub z_far: f32,
    /// 水平旋转角（弧度），围绕角色 Y 轴
    pub yaw: f32,
    /// 俯仰角（弧度），向上为正
    pub pitch: f32,
    /// 摄像机到角色的距离
    pub distance: f32,
    /// 注视点相对角色脚底的高度（围绕/看向角色身体中心）
    pub target_height: f32,
    /// 跟随平滑系数（0=即时跟随，1=不平滑）
    pub follow_smooth: f32,
    /// 鼠标旋转灵敏度
    pub orbit_sensitivity: f32,
    /// 缩放灵敏度
    pub zoom_sensitivity: f32,
    /// 最小俯仰角
    pub min_pitch: f32,
    /// 最大俯仰角
    pub max_pitch: f32,
    /// 最小距离
    pub min_distance: f32,
    /// 最大距离
    pub max_distance: f32,
    /// 平滑后的摄像机目标位置（用于插值）
    target_position: Point3<f32>,
}

impl ThirdPersonCamera {
    /// 使用项目配置注入的参数创建摄像机（由 Geese Launcher 生成）。
    pub fn new() -> Self {
        let fov: f32 = {{camera_fov}};
        let z_near: f32 = {{camera_z_near}};
        let z_far: f32 = {{camera_z_far}};
        let yaw: f32 = {{camera_yaw}};
        let pitch: f32 = {{camera_pitch}};
        let distance: f32 = {{camera_distance}};
        let target_height: f32 = {{camera_eye_height}};
        let follow_smooth: f32 = {{camera_follow_smooth}};
        let orbit_sensitivity: f32 = {{player_mouse_sensitivity}};

        // 初始位置 = 角色位于原点时的理想轨道位置
        let cos_pitch = pitch.cos();
        let default_pos = Point3::new(
            distance * yaw.sin() * cos_pitch,
            target_height + distance * pitch.sin(),
            distance * yaw.cos() * cos_pitch,
        );

        Self {
            position: default_pos,
            fov,
            z_near,
            z_far,
            yaw,
            pitch,
            distance,
            target_height,
            follow_smooth,
            orbit_sensitivity,
            zoom_sensitivity: 5.0,
            min_pitch: -0.3,
            max_pitch: 1.2,
            min_distance: 2.0,
            max_distance: 30.0,
            target_position: default_pos,
        }
    }

    /// 计算摄像机在球坐标系下的理想位置（围绕角色身体中心）。
    fn compute_ideal_position(&self, player_pos: Point3<f32>) -> Point3<f32> {
        let center = Point3::new(
            player_pos.x,
            player_pos.y + self.target_height,
            player_pos.z,
        );
        let cos_pitch = self.pitch.cos();
        let x = self.distance * self.yaw.sin() * cos_pitch;
        let y = self.distance * self.pitch.sin();
        let z = self.distance * self.yaw.cos() * cos_pitch;

        Point3::new(center.x + x, center.y + y, center.z + z)
    }

    /// 更新摄像机，平滑跟随角色。
    /// `player_pos`: 角色脚底位置。
    pub fn update(&mut self, player_pos: Point3<f32>, dt: f32) {
        self.target_position = self.compute_ideal_position(player_pos);

        // 帧率无关的指数平滑：follow_smooth 视为 60 FPS 基准下的每帧插值系数
        let smooth = self.follow_smooth.clamp(0.0, 1.0);
        let t = 1.0 - (1.0 - smooth).powf(dt * 60.0);
        self.position.x += (self.target_position.x - self.position.x) * t;
        self.position.y += (self.target_position.y - self.position.y) * t;
        self.position.z += (self.target_position.z - self.position.z) * t;
    }

    /// 处理鼠标拖拽旋转摄像机。
    /// `mouse_delta`: 两帧间的鼠标位移（像素）。
    pub fn orbit(&mut self, mouse_delta: (f32, f32)) {
        self.yaw -= mouse_delta.0 * self.orbit_sensitivity;
        self.pitch += mouse_delta.1 * self.orbit_sensitivity;
        self.pitch = self.pitch.clamp(self.min_pitch, self.max_pitch);
        // 规范化 yaw 到 [-π, π]，避免长时间旋转后的浮点精度损失
        self.yaw = (self.yaw + std::f32::consts::PI).rem_euclid(2.0 * std::f32::consts::PI)
            - std::f32::consts::PI;
    }

    /// 处理滚轮缩放。
    /// `scroll_delta`: 正值放大（拉近），负值缩小（拉远）。
    pub fn zoom(&mut self, scroll_delta: f32) {
        self.distance -= scroll_delta * self.zoom_sensitivity;
        self.distance = self.distance.clamp(self.min_distance, self.max_distance);
    }

    /// 上方向量。
    pub fn up(&self) -> Vector3<f32> {
        Vector3::new(0.0, 1.0, 0.0)
    }

    /// 获取摄像机在 XZ 平面上的前向方向（用于角色移动）。
    pub fn forward_xz(&self) -> Vector3<f32> {
        Vector3::new(-self.yaw.sin(), 0.0, -self.yaw.cos())
    }

    /// 获取摄像机在 XZ 平面上的右向方向（用于角色移动）。
    pub fn right_xz(&self) -> Vector3<f32> {
        Vector3::new(self.yaw.cos(), 0.0, -self.yaw.sin())
    }

    /// 构建视图矩阵，看向角色身体中心。
    pub fn view_matrix(&self, player_pos: Point3<f32>) -> Matrix4<f32> {
        let look_target = Point3::new(
            player_pos.x,
            player_pos.y + self.target_height,
            player_pos.z,
        );
        Matrix4::look_at_rh(self.position, look_target, self.up())
    }

    /// 构建透视投影矩阵。
    pub fn projection_matrix(&self, aspect_ratio: f32) -> Matrix4<f32> {
        cgmath::perspective(
            Rad(self.fov.to_radians()),
            aspect_ratio,
            self.z_near,
            self.z_far,
        )
    }

    /// 构建 view-projection 矩阵。
    pub fn view_projection(&self, player_pos: Point3<f32>, aspect_ratio: f32) -> Matrix4<f32> {
        self.projection_matrix(aspect_ratio) * self.view_matrix(player_pos)
    }
}
